class GameEngine { constructor() { this.character = null; this.currentLocation = null; this.battleLog = document.getElementById('battle-log'); this.locationDescription = document.getElementById('location-description'); this.inventory = document.getElementById('inventory'); this.quests = document.getElementById('quests'); // Инициализация кнопок this.initButtons(); // Загрузка сохраненной игры this.loadGame(); } initButtons() { document.getElementById('explore-btn').addEventListener('click', () => this.explore()); document.getElementById('rest-btn').addEventListener('click', () => this.rest()); document.getElementById('shop-btn').addEventListener('click', () => this.openShop()); document.getElementById('quests-btn').addEventListener('click', () => this.showQuests()); } loadGame() { // Загрузка сохраненной игры или создание новой const savedGame = localStorage.getItem('eternalQuest'); if (savedGame) { const gameData = JSON.parse(savedGame); this.character = new Character(gameData.character.name, gameData.character.class); Object.assign(this.character, gameData.character); this.currentLocation = gameData.currentLocation; } else { this.startNewGame(); } this.updateUI(); } startNewGame() { // Создание нового персонажа this.character = new Character('Герой', 'warrior'); this.currentLocation = 'village'; // Добавление начальных предметов this.character.addItem(items.weapons[0]); this.character.addItem(items.armor[0]); // Добавление начального квеста this.character.addQuest(quests.forest[0]); this.saveGame(); } saveGame() { const gameData = { character: this.character, currentLocation: this.currentLocation }; localStorage.setItem('eternalQuest', JSON.stringify(gameData)); } updateUI() { // Обновление характеристик document.getElementById('level').textContent = this.character.level; document.getElementById('exp').textContent = this.character.exp; document.getElementById('exp-to-next').textContent = this.character.expToNextLevel; document.getElementById('hp').textContent = this.character.stats.hp; document.getElementById('max-hp').textContent = this.character.stats.maxHp; document.getElementById('mp').textContent = this.character.stats.mp; document.getElementById('max-mp').textContent = this.character.stats.maxMp; document.getElementById('str').textContent = this.character.stats.str; document.getElementById('dex').textContent = this.character.stats.dex; document.getElementById('int').textContent = this.character.stats.int; // Обновление полосок здоровья и маны document.getElementById('hp-bar').style.width = `${(this.character.stats.hp / this.character.stats.maxHp) * 100}%`; document.getElementById('mp-bar').style.width = `${(this.character.stats.mp / this.character.stats.maxMp) * 100}%`; // Обновление инвентаря this.updateInventory(); // Обновление квестов this.updateQuests(); // Обновление описания локации this.updateLocationDescription(); } updateInventory() { this.inventory.innerHTML = ''; // Отображение оружия this.character.inventory.weapons.forEach(weapon => { const itemElement = document.createElement('div'); itemElement.className = 'item'; itemElement.textContent = weapon.name; itemElement.addEventListener('click', () => this.equipItem(weapon)); this.inventory.appendChild(itemElement); }); // Отображение брони this.character.inventory.armor.forEach(armor => { const itemElement = document.createElement('div'); itemElement.className = 'item'; itemElement.textContent = armor.name; itemElement.addEventListener('click', () => this.equipItem(armor)); this.inventory.appendChild(itemElement); }); // Отображение зелий this.character.inventory.potions.forEach(potion => { const itemElement = document.createElement('div'); itemElement.className = 'item'; itemElement.textContent = potion.name; itemElement.addEventListener('click', () => this.usePotion(potion)); this.inventory.appendChild(itemElement); }); } updateQuests() { this.quests.innerHTML = ''; this.character.status.quests.forEach(quest => { const questElement = document.createElement('div'); questElement.className = 'quest-item'; questElement.innerHTML = `

${quest.title}

${quest.description}

Награда: ${quest.reward.exp} опыта, ${quest.reward.gold} золота

`; this.quests.appendChild(questElement); }); } updateLocationDescription() { const location = locations[this.currentLocation]; this.locationDescription.textContent = location.description; } explore() { const location = locations[this.currentLocation]; if (location.monsters.length > 0) { const randomMonster = location.monsters[Math.floor(Math.random() * location.monsters.length)]; this.startBattle(randomMonster); } else { this.addToLog('Здесь нечего исследовать.'); } } startBattle(monster) { this.addToLog(`Вы встретили ${monster.name}!`); while (this.character.stats.hp > 0 && monster.stats.hp > 0) { // Атака персонажа const playerDamage = this.character.attack(monster); this.addToLog(`Вы нанесли ${playerDamage} урона ${monster.name}!`); if (monster.stats.hp <= 0) { this.addToLog(`Вы победили ${monster.name}!`); this.character.gainExp(monster.expReward); break; } // Атака монстра const monsterDamage = monster.attack(this.character); this.addToLog(`${monster.name} нанес вам ${monsterDamage} урона!`); if (this.character.stats.hp <= 0) { this.addToLog('Вы погибли!'); this.character.die(); break; } } this.updateUI(); this.saveGame(); } rest() { this.character.stats.hp = this.character.stats.maxHp; this.character.stats.mp = this.character.stats.maxMp; this.addToLog('Вы отдохнули и восстановили здоровье и ману.'); this.updateUI(); this.saveGame(); } openShop() { // TODO: Реализовать магазин this.addToLog('Магазин пока закрыт.'); } showQuests() { // TODO: Реализовать меню квестов this.addToLog('Меню квестов пока недоступно.'); } equipItem(item) { this.character.equipItem(item); this.addToLog(`Вы экипировали ${item.name}`); this.updateUI(); this.saveGame(); } usePotion(potion) { if (potion.heal) { this.character.stats.hp = Math.min(this.character.stats.maxHp, this.character.stats.hp + potion.heal); this.addToLog(`Вы использовали ${potion.name} и восстановили ${potion.heal} здоровья.`); } else if (potion.restoreMp) { this.character.stats.mp = Math.min(this.character.stats.maxMp, this.character.stats.mp + potion.restoreMp); this.addToLog(`Вы использовали ${potion.name} и восстановили ${potion.restoreMp} маны.`); } // Удаление использованного зелья this.character.inventory.potions = this.character.inventory.potions.filter(p => p !== potion); this.updateUI(); this.saveGame(); } addToLog(message) { const logEntry = document.createElement('p'); logEntry.textContent = message; this.battleLog.appendChild(logEntry); this.battleLog.scrollTop = this.battleLog.scrollHeight; } } // Инициализация игры при загрузке страницы window.addEventListener('load', () => { const game = new GameEngine(); });
